var bot = BotManager.getCurrentBot();

function log(message) {
  recordDiagnostic("log", message, false);
  if (GATEWAY_CONFIG.enableConsoleLog === true) {
    java.lang.System.out.println("[bot_gateway] " + message);
  }
}

function validateGatewayConfig() {
  if (!GATEWAY_CONFIG.botSecret || GATEWAY_CONFIG.botSecret === "change-me") {
    throw new Error("GATEWAY_CONFIG.botSecret must be replaced before startup");
  }
  if (!GATEWAY_CONFIG.socketToken || GATEWAY_CONFIG.socketToken === "change-me-too") {
    throw new Error("GATEWAY_CONFIG.socketToken must be replaced before startup");
  }
  if (!GATEWAY_CONFIG.adminRoomName || GATEWAY_CONFIG.adminRoomName === "change-me-admin-room") {
    throw new Error("GATEWAY_CONFIG.adminRoomName must be replaced before startup");
  }
  if (!GATEWAY_CONFIG.deliveryAckUrl) {
    throw new Error("GATEWAY_CONFIG.deliveryAckUrl must not be empty");
  }
  if (!GATEWAY_CONFIG.controlAuthorName) {
    throw new Error("GATEWAY_CONFIG.controlAuthorName must not be empty");
  }
  if (!GATEWAY_CONFIG.controlRoomName) {
    throw new Error("GATEWAY_CONFIG.controlRoomName must not be empty");
  }
}

function nowIsoString() {
  var format = new java.text.SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ssXXX");
  format.setTimeZone(java.util.TimeZone.getTimeZone("Asia/Seoul"));
  return format.format(new java.util.Date());
}

function nowMillis() {
  return new java.util.Date().getTime();
}

function sleepMs(ms) {
  java.lang.Thread.sleep(ms);
}

function truncateText(value, limit) {
  var text = String(value == null ? "" : value);
  var max = Number(limit || 500);
  if (text.length <= max) {
    return text;
  }
  return text.substring(0, max) + "...";
}

function initDiagnosticState() {
  diagState = {
    started_at: nowIsoString(),
    started_ms: nowMillis(),
    uptime_ms: 0,
    gateway_ready: false,
    sequence: 0,
    inbound_received: 0,
    inbound_enqueued: 0,
    inbound_delivered: 0,
    inbound_spooled: 0,
    inbound_queue_full: 0,
    webhook_error_count: 0,
    ack_ok_count: 0,
    ack_error_count: 0,
    pending_flush_count: 0,
    pending_flush_error_count: 0,
    worker_error_count: 0,
    pending_loaded_count: 0,
    pending_after_flush_count: 0,
    queue_size: 0,
    last_event_at: null,
    last_inbound_at: null,
    last_webhook_ok_at: null,
    last_ack_ok_at: null,
    last_error_at: null,
    last_error_type: null,
    last_error: null,
    recent: []
  };
}

function refreshDiagnosticSnapshot() {
  if (diagState == null) {
    return;
  }
  diagState.uptime_ms = nowMillis() - Number(diagState.started_ms || nowMillis());
  diagState.gateway_ready = gatewayReady === true;
  try {
    diagState.queue_size = inboundQueue.size();
  } catch (ignore) {}
}

function recordDiagnostic(type, detail, isError) {
  if (diagState == null) {
    return;
  }
  var currentMs = nowMillis();
  refreshDiagnosticSnapshot();
  diagState.sequence += 1;
  diagState.last_event_at = nowIsoString();
  var entry = {
    seq: diagState.sequence,
    at: diagState.last_event_at,
    type: String(type || "event"),
    detail: truncateText(detail, 500)
  };
  diagState.recent.push(entry);
  while (diagState.recent.length > Number(GATEWAY_CONFIG.diagnosticRecentLimit || 30)) {
    diagState.recent.shift();
  }
  if (isError === true) {
    diagState.last_error_at = entry.at;
    diagState.last_error_type = entry.type;
    diagState.last_error = entry.detail;
  }

  var errorInterval = Number(GATEWAY_CONFIG.diagnosticErrorPersistMinIntervalMs || 10000);
  var force = isError === true && currentMs - lastDiagnosticPersistMs >= errorInterval;
  persistDiagnostic(force);
}

function persistDiagnostic(force) {
  if (diagState == null) {
    return;
  }
  var currentMs = nowMillis();
  var interval = Number(GATEWAY_CONFIG.diagnosticPersistIntervalMs || 60000);
  if (force !== true && currentMs - lastDiagnosticPersistMs < interval) {
    return;
  }
  refreshDiagnosticSnapshot();
  lastDiagnosticPersistMs = currentMs;
  try {
    Database.writeObject(GATEWAY_CONFIG.diagnosticFile, diagState);
  } catch (ignore) {}
}

function countPendingEvents() {
  try {
    return loadPendingEvents().length;
  } catch (ignore) {
    return -1;
  }
}

initDiagnosticState();

function readObjectFile(path, emptyValue) {
  try {
    if (!Database.exists(path)) {
      return emptyValue;
    }
    var payload = Database.readObject(path);
    return payload == null ? emptyValue : payload;
  } catch (error) {
    recordDiagnostic("database_read_error", path + ": " + error, true);
    return emptyValue;
  }
}

function writeObjectFile(path, payload) {
  try {
    Database.writeObject(path, payload);
    return true;
  } catch (error) {
    if (String(path) !== String(GATEWAY_CONFIG.diagnosticFile)) {
      recordDiagnostic("database_write_error", path + ": " + error, true);
    }
    return false;
  }
}

function loadProcessedIds() {
  var payload = readObjectFile(GATEWAY_CONFIG.dedupeFile, { ids: {} });
  if (payload.ids == null) {
    return {};
  }
  return payload.ids;
}

function saveProcessedIds(ids) {
  var keys = Object.keys(ids);
  if (keys.length > GATEWAY_CONFIG.dedupeLimit) {
    keys.sort(function(a, b) {
      return ids[b] - ids[a];
    });
    var trimmed = {};
    for (var i = 0; i < GATEWAY_CONFIG.dedupeLimit; i++) {
      trimmed[keys[i]] = ids[keys[i]];
    }
    ids = trimmed;
  }
  writeObjectFile(GATEWAY_CONFIG.dedupeFile, { ids: ids });
}

function hasProcessedMessage(messageId) {
  var ids = loadProcessedIds();
  return ids[String(messageId)] != null;
}

function rememberMessage(messageId) {
  var ids = loadProcessedIds();
  ids[String(messageId)] = nowMillis();
  saveProcessedIds(ids);
}

function buildFallbackLogId(message) {
  var stableTimestamp = "";
  if (message.timestamp != null) stableTimestamp = String(message.timestamp);
  else if (message.time != null) stableTimestamp = String(message.time);
  else if (message.createdAt != null) stableTimestamp = String(message.createdAt);

  var seed = [
    String(message.room || ""),
    String(message.content || ""),
    String(message.packageName || ""),
    message.author ? String(message.author.name || "") : "",
    stableTimestamp
  ].join("|");
  return Security.sha256(seed);
}

function extractLogId(message) {
  if (message.logId != null) {
    return String(message.logId);
  }
  return buildFallbackLogId(message);
}

function loadPendingEvents() {
  var payload = readObjectFile(GATEWAY_CONFIG.pendingQueueFile, { events: [] });
  if (payload.events == null) {
    return [];
  }
  return payload.events;
}

function savePendingEvents(events) {
  if (events.length > GATEWAY_CONFIG.pendingQueueLimit) {
    events.sort(function(a, b) {
      return Number(a.enqueued_at || 0) - Number(b.enqueued_at || 0);
    });
    events = events.slice(events.length - GATEWAY_CONFIG.pendingQueueLimit);
  }
  writeObjectFile(GATEWAY_CONFIG.pendingQueueFile, { events: events });
}

function upsertPendingEvent(payload, errorText) {
  var events = loadPendingEvents();
  var logId = String(payload.logId || "");
  var updated = false;
  for (var i = 0; i < events.length; i++) {
    if (String(events[i].payload.logId || "") === logId) {
      events[i].retry_count = Number(events[i].retry_count || 0) + 1;
      events[i].last_error = String(errorText || "");
      events[i].next_attempt_at = nowMillis() + retryDelayMs(events[i].retry_count);
      updated = true;
      break;
    }
  }
  if (!updated) {
    events.push({
      payload: payload,
      retry_count: 0,
      enqueued_at: nowMillis(),
      next_attempt_at: nowMillis(),
      last_error: String(errorText || "")
    });
  }
  savePendingEvents(events);
}

function retryDelayMs(retryCount) {
  var exponent = Math.min(Number(retryCount || 0), 5);
  return GATEWAY_CONFIG.webhookRetryBaseDelayMs * Math.pow(2, exponent);
}

function postJson(urlText, payload, headers) {
  var url = new java.net.URL(urlText);
  var connection = url.openConnection();
  connection.setRequestMethod("POST");
  connection.setDoOutput(true);
  connection.setConnectTimeout(Number(GATEWAY_CONFIG.httpConnectTimeoutMs || 5000));
  connection.setReadTimeout(Number(GATEWAY_CONFIG.httpReadTimeoutMs || 10000));
  connection.setRequestProperty("Content-Type", "application/json; charset=UTF-8");
  if (headers != null) {
    for (var key in headers) {
      connection.setRequestProperty(String(key), String(headers[key]));
    }
  }

  var output = new java.io.OutputStreamWriter(connection.getOutputStream(), "UTF-8");
  output.write(gson.toJson(payload));
  output.flush();
  output.close();

  var code = connection.getResponseCode();
  var stream = code >= 200 && code < 400 ? connection.getInputStream() : connection.getErrorStream();
  var responseBuilder = new java.lang.StringBuilder();
  if (stream != null) {
    var reader = new java.io.BufferedReader(new java.io.InputStreamReader(stream, "UTF-8"));
    var line;
    while ((line = reader.readLine()) != null) {
      responseBuilder.append(line);
    }
    reader.close();
  }
  try {
    connection.disconnect();
  } catch (ignore) {}
  return { code: code, body: String(responseBuilder.toString()) };
}

function deliverInboundPayload(payload) {
  var lastError = "";
  for (var attempt = 0; attempt < GATEWAY_CONFIG.webhookRetryCount; attempt++) {
    try {
      var result = postJson(GATEWAY_CONFIG.webhookUrl, payload, {
        "X-Bot-Secret": GATEWAY_CONFIG.botSecret
      });
      if (result.code >= 200 && result.code < 300) {
        if (diagState != null) {
          diagState.inbound_delivered += 1;
          diagState.last_webhook_ok_at = nowIsoString();
          persistDiagnostic(false);
        }
        return { ok: true, error: "" };
      }
      lastError = "webhook failed: " + result.code + " " + result.body;
      if (diagState != null) {
        diagState.webhook_error_count += 1;
      }
      recordDiagnostic("webhook_error", lastError, true);
    } catch (error) {
      lastError = "webhook exception: " + error;
      if (diagState != null) {
        diagState.webhook_error_count += 1;
      }
      recordDiagnostic("webhook_exception", lastError, true);
    }
    if (attempt + 1 < GATEWAY_CONFIG.webhookRetryCount) {
      sleepMs(GATEWAY_CONFIG.webhookRetryBaseDelayMs * (attempt + 1));
    }
  }
  return { ok: false, error: lastError };
}

function flushPendingEvents() {
  var events = loadPendingEvents();
  if (diagState != null) {
    diagState.pending_loaded_count = events.length;
  }
  if (events.length === 0) {
    return;
  }
  var remaining = [];
  var processed = 0;
  var currentMs = nowMillis();
  for (var i = 0; i < events.length; i++) {
    var entry = events[i];
    if (processed >= GATEWAY_CONFIG.pendingFlushBatchSize) {
      remaining.push(entry);
      continue;
    }
    if (Number(entry.next_attempt_at || 0) > currentMs) {
      remaining.push(entry);
      continue;
    }
    processed += 1;
    var result = deliverInboundPayload(entry.payload);
    if (!result.ok) {
      entry.retry_count = Number(entry.retry_count || 0) + 1;
      entry.last_error = result.error;
      entry.next_attempt_at = nowMillis() + retryDelayMs(entry.retry_count);
      remaining.push(entry);
    }
  }
  savePendingEvents(remaining);
  if (diagState != null) {
    diagState.pending_flush_count += 1;
    diagState.pending_after_flush_count = remaining.length;
    persistDiagnostic(false);
  }
}

function processInboundPayload(payload) {
  var result = deliverInboundPayload(payload);
  if (!result.ok) {
    if (diagState != null) {
      diagState.inbound_spooled += 1;
    }
    upsertPendingEvent(payload, result.error);
  }
}

function processQueuedInboundPayload(payloadText) {
  var payload = JSON.parse(String(payloadText));
  processInboundPayload(payload);
}

function enqueueInboundPayload(payload) {
  var payloadText = gson.toJson(payload);
  var accepted = inboundQueue.offer(payloadText);
  if (diagState != null) {
    diagState.inbound_received += 1;
    diagState.last_inbound_at = nowIsoString();
    diagState.queue_size = inboundQueue.size();
    if (accepted) {
      diagState.inbound_enqueued += 1;
    } else {
      diagState.inbound_queue_full += 1;
    }
  }
  if (!accepted) {
    recordDiagnostic("inbound_queue_full", "logId=" + String(payload.logId || ""), true);
    upsertPendingEvent(payload, "inbound queue full");
  } else {
    persistDiagnostic(false);
  }
  return accepted;
}

function startPendingFlushWorker() {
  if (!gatewayReady) {
    log("gateway disabled: pending flush worker not started");
    return;
  }
  if (pendingFlushThread != null) {
    return;
  }

  pendingFlushWorkerToken += 1;
  var workerToken = pendingFlushWorkerToken;
  pendingFlushThread = new java.lang.Thread(new java.lang.Runnable({
    run: function() {
      recordDiagnostic("worker_started", "gateway worker started", false);
      var nextFlushAt = nowMillis();
      while (workerToken === pendingFlushWorkerToken) {
        try {
          var currentMs = nowMillis();
          if (currentMs >= nextFlushAt) {
            flushPendingEvents();
            nextFlushAt = nowMillis() + Number(GATEWAY_CONFIG.pendingFlushIntervalMs || 60000);
          }
          var payloadText = inboundQueue.poll(1000, java.util.concurrent.TimeUnit.MILLISECONDS);
          if (payloadText != null) {
            processQueuedInboundPayload(payloadText);
          }
          persistDiagnostic(false);
        } catch (error) {
          if (diagState != null) {
            diagState.worker_error_count += 1;
          }
          recordDiagnostic("worker_error", String(error), true);
          sleepMs(1000);
        }
      }
      recordDiagnostic("worker_stopped", "gateway worker stopped", false);
      persistDiagnostic(true);
    }
  }));
  try {
    pendingFlushThread.setName("bot_gateway_worker");
  } catch (ignore2) {}
  pendingFlushThread.start();
}

function stopPendingFlushWorker() {
  if (pendingFlushThread == null) {
    return;
  }
  pendingFlushWorkerToken += 1;
  pendingFlushThread = null;
  recordDiagnostic("worker_stop_requested", "gateway worker stop requested", false);
  persistDiagnostic(true);
}

function sendAdminReport(text) {
  try {
    if (!bot.canReply(GATEWAY_CONFIG.adminRoomName, GATEWAY_CONFIG.packageName)) {
      log("admin room is not replyable: " + text);
      return;
    }
    bot.send(GATEWAY_CONFIG.adminRoomName, text, GATEWAY_CONFIG.packageName);
  } catch (error) {
    log("admin report failed: " + error + " / original=" + text);
  }
}

function sendDeliveryAck(messageId, roomName, packageName, status, errorCode, errorMessage) {
  var payload = {
    message_id: String(messageId || ""),
    status: String(status || "retryable_error"),
    target_room: String(roomName || ""),
    package_name: String(packageName || GATEWAY_CONFIG.packageName),
    error_code: errorCode == null ? null : String(errorCode),
    error_message: errorMessage == null ? null : String(errorMessage)
  };
  try {
    var result = postJson(GATEWAY_CONFIG.deliveryAckUrl, payload, {
      "X-Bot-Secret": GATEWAY_CONFIG.botSecret
    });
    if (result.code >= 200 && result.code < 300) {
      if (diagState != null) {
        diagState.ack_ok_count += 1;
        diagState.last_ack_ok_at = nowIsoString();
        persistDiagnostic(false);
      }
      return true;
    }
    if (diagState != null) {
      diagState.ack_error_count += 1;
    }
    recordDiagnostic("delivery_ack_error", "message_id=" + payload.message_id + ": " + result.code + " " + result.body, true);
  } catch (error) {
    if (diagState != null) {
      diagState.ack_error_count += 1;
    }
    recordDiagnostic("delivery_ack_exception", "message_id=" + payload.message_id + ": " + error, true);
  }
  return false;
}

function buildDiagnosticReport() {
  refreshDiagnosticSnapshot();
  var pendingCount = countPendingEvents();
  var lines = [
    "[gateway_diag]",
    "ready=" + String(gatewayReady === true),
    "started_at=" + String(diagState.started_at || ""),
    "uptime_ms=" + String(diagState.uptime_ms || 0),
    "queue_size=" + String(diagState.queue_size || 0),
    "pending_count=" + String(pendingCount),
    "inbound_received=" + String(diagState.inbound_received || 0),
    "inbound_enqueued=" + String(diagState.inbound_enqueued || 0),
    "inbound_delivered=" + String(diagState.inbound_delivered || 0),
    "inbound_spooled=" + String(diagState.inbound_spooled || 0),
    "webhook_error_count=" + String(diagState.webhook_error_count || 0),
    "ack_error_count=" + String(diagState.ack_error_count || 0),
    "worker_error_count=" + String(diagState.worker_error_count || 0),
    "last_webhook_ok_at=" + String(diagState.last_webhook_ok_at || ""),
    "last_ack_ok_at=" + String(diagState.last_ack_ok_at || ""),
    "last_error_type=" + String(diagState.last_error_type || ""),
    "last_error_at=" + String(diagState.last_error_at || ""),
    "last_error=" + truncateText(diagState.last_error || "", 300)
  ];
  return lines.join("\n");
}

function handleLocalDiagnosticCommand(message) {
  if (message == null) {
    return false;
  }
  if (String(message.room || "") !== String(GATEWAY_CONFIG.adminRoomName || "")) {
    return false;
  }
  var content = String(message.content || "").replace(/^\s+|\s+$/g, "");
  if (content !== "@게이트웨이진단" && content !== "@gateway_diag") {
    return false;
  }
  persistDiagnostic(true);
  sendAdminReport(buildDiagnosticReport());
  return true;
}

function isControlMessage(message) {
  if (message == null) {
    return false;
  }
  if (message.isDebugRoom !== true) {
    return false;
  }
  var authorName = message.author ? String(message.author.name || "") : "";
  if (authorName !== String(GATEWAY_CONFIG.controlAuthorName || "")) {
    return false;
  }
  return String(message.room || "") === String(GATEWAY_CONFIG.controlRoomName || "");
}

function handleControlMessage(message) {
  if (!isControlMessage(message)) {
    return false;
  }

  var content = String(message.content || "");
  var command = null;
  try {
    command = JSON.parse(content);
  } catch (error) {
    sendAdminReport("[gateway][control] malformed JSON ignored: " + error);
    return true;
  }

  if (String(command.token || "") !== GATEWAY_CONFIG.socketToken) {
    sendAdminReport("[gateway][control] invalid token for message_id=" + String(command.message_id || ""));
    return true;
  }

  if (String(command.type || "") !== "send_message") {
    sendAdminReport("[gateway][control] unsupported type=" + String(command.type || ""));
    return true;
  }

  var messageId = String(command.message_id || "");
  if (!messageId) {
    sendAdminReport("[gateway][control] missing message_id");
    return true;
  }

  var roomName = String(command.target_room || "");
  var text = String(command.text || "");
  var packageName = String(command.package_name || GATEWAY_CONFIG.packageName);
  if (!roomName || !text) {
    sendAdminReport("[gateway][control] invalid payload for message_id=" + messageId);
    sendDeliveryAck(messageId, roomName, packageName, "fatal_error", "gateway_invalid_payload", "invalid target_room or text");
    return true;
  }

  if (hasProcessedMessage(messageId)) {
    log("duplicate control message acknowledged: " + messageId);
    sendDeliveryAck(messageId, roomName, packageName, "ok", null, null);
    return true;
  }

  try {
    if (!bot.canReply(roomName, packageName)) {
      log("[gateway][delivery] cannot reply to room=" + roomName + " message_id=" + messageId);
      sendDeliveryAck(
        messageId,
        roomName,
        packageName,
        "retryable_error",
        "gateway_cannot_reply",
        "bot.canReply returned false"
      );
      return true;
    }

    var success = bot.send(roomName, text, packageName);
    if (!success) {
      log("[gateway][delivery] bot.send returned false for room=" + roomName + " message_id=" + messageId);
      sendDeliveryAck(
        messageId,
        roomName,
        packageName,
        "retryable_error",
        "gateway_send_false",
        "bot.send returned false"
      );
      return true;
    }

    rememberMessage(messageId);
    sendDeliveryAck(messageId, roomName, packageName, "ok", null, null);
  } catch (error2) {
    log("[gateway][delivery] exception for room=" + roomName + " message_id=" + messageId + ": " + String(error2));
    sendDeliveryAck(
      messageId,
      roomName,
      packageName,
      "retryable_error",
      "gateway_exception",
      String(error2)
    );
  }
  return true;
}

function forwardInbound(message) {
  if (!gatewayReady) {
    log("gateway disabled: inbound forwarding skipped");
    return;
  }
  var payload = {
    room: String(message.room || ""),
    content: String(message.content || ""),
    logId: extractLogId(message),
    packageName: String(message.packageName || GATEWAY_CONFIG.packageName),
    author: {
      name: message.author ? String(message.author.name || "") : null,
      hash: null
    },
    client_received_at: nowIsoString()
  };

  enqueueInboundPayload(payload);
}

try {
  bot.removeAllListeners(Event.MESSAGE);
  bot.removeAllListeners(Event.START_COMPILE);
} catch (ignore) {}

bot.on(Event.MESSAGE, function(message) {
  if (handleControlMessage(message)) {
    return;
  }
  if (handleLocalDiagnosticCommand(message)) {
    return;
  }
  forwardInbound(message);
});

bot.on(Event.START_COMPILE, function() {
  stopPendingFlushWorker();
});

try {
  validateGatewayConfig();
  gatewayReady = true;
  recordDiagnostic("gateway_started", "gateway startup completed", false);
  startPendingFlushWorker();
  persistDiagnostic(true);
} catch (error) {
  gatewayReady = false;
  recordDiagnostic("gateway_startup_blocked", String(error), true);
}
