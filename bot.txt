function getScriptGlobalScope() {
  try {
    return (function() {
      return this;
    })();
  } catch (e) {
    return null;
  }
}

function getRuntimeRegistry() {
  var registryKey = "__PPIRAE_RUNTIME_REGISTRY__";
  var holder = null;
  var registry = null;

  try {
    holder = getScriptGlobalScope();
    if (holder && holder[registryKey]) {
      return holder[registryKey];
    }
  } catch (e) {}

  try {
    if (bot && bot[registryKey]) {
      return bot[registryKey];
    }
  } catch (e2) {}

  registry = {};

  try {
    if (holder) {
      holder[registryKey] = registry;
      return registry;
    }
  } catch (e3) {}

  try {
    if (bot) {
      bot[registryKey] = registry;
      return registry;
    }
  } catch (e4) {}

  return registry;
}

function getOrCreateRuntimeSlot(registry, key) {
  var normalizedKey = trimString(key) || "default";
  if (!registry[normalizedKey] || typeof registry[normalizedKey] !== "object") {
    registry[normalizedKey] = {
      startupCount: 0,
      current: null
    };
  }
  return registry[normalizedKey];
}

function generateInstanceId() {
  var randomPart = "0";
  try {
    randomPart = java.lang.Long.toHexString(
      java.lang.Double.doubleToLongBits(java.lang.Math.random())
    );
  } catch (e) {}
  return String(nowMillis()) + "-" + String(randomPart);
}

function safeString(value) {
  if (value === null || value === undefined) {
    return "";
  }
  return String(value);
}

function trimString(value) {
  return safeString(value).replace(/^\s+|\s+$/g, "");
}

function isArray(value) {
  return Object.prototype.toString.call(value) === "[object Array]";
}

function countObjectKeys(value) {
  var count = 0;
  var key;
  if (!value || typeof value !== "object") {
    return 0;
  }
  for (key in value) {
    if (value.hasOwnProperty(key)) {
      count += 1;
    }
  }
  return count;
}

function runtimeMemoryInfo() {
  var runtime;
  var total;
  var free;
  var max;
  var used;

  try {
    runtime = java.lang.Runtime.getRuntime();
    total = runtime.totalMemory();
    free = runtime.freeMemory();
    max = runtime.maxMemory();
    used = total - free;
    return {
      used: used,
      free: free,
      total: total,
      max: max
    };
  } catch (e) {
    return null;
  }
}

function formatBytes(value) {
  var bytes = parseInt(value, 10);
  if (!bytes && bytes !== 0) {
    return "-";
  }
  return String(Math.round((bytes / 1024 / 1024) * 10) / 10) + "MB";
}

function buildMemoryText() {
  var info = runtimeMemoryInfo();
  if (!info) {
    return "memory: unavailable";
  }
  return (
    "memoryUsed: " + formatBytes(info.used) + "\n" +
    "memoryFree: " + formatBytes(info.free) + "\n" +
    "memoryTotal: " + formatBytes(info.total) + "\n" +
    "memoryMax: " + formatBytes(info.max)
  );
}

function toPlainArray(value) {
  var result = [];
  var i;
  var size;

  if (!value) {
    return result;
  }

  if (isArray(value)) {
    return value.slice(0);
  }

  try {
    if (typeof value.length === "number") {
      for (i = 0; i < value.length; i++) {
        result.push(value[i]);
      }
      return result;
    }
  } catch (e) {}

  try {
    if (typeof value.size === "function" && typeof value.get === "function") {
      size = value.size();
      for (i = 0; i < size; i++) {
        result.push(value.get(i));
      }
      return result;
    }
  } catch (e2) {}

  return result;
}

function previewText(text, limit) {
  var value = safeString(text);
  var maxLength = limit || 120;
  if (value.length <= maxLength) {
    return value;
  }
  return value.substring(0, maxLength) + "...";
}

function nowIsoString() {
  var sdf = new java.text.SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ssXXX");
  return String(sdf.format(new java.util.Date()));
}

function nowMillis() {
  return java.lang.System.currentTimeMillis();
}

function logThrottled(level, throttleKey, message, intervalMs) {
  var key = trimString(throttleKey) || "default";
  var now = nowMillis();
  var interval = intervalMs || CONFIG.DIAGNOSTIC_LOG_THROTTLE_MS;
  var lastAt = STATE.logThrottleAtMs[key] || 0;

  if (lastAt && now - lastAt < interval) {
    return false;
  }

  STATE.logThrottleAtMs[key] = now;
  if (level === "error") {
    Log.e(message);
  } else {
    Log.i(message);
  }
  return true;
}

function notifyAdminDiagnosticThrottled(kind, detail) {
  var normalizedKind = trimString(kind) || "diagnostic";
  var text;
  var key = "__diagnostic_global__";
  var now = nowMillis();
  var interval = CONFIG.ADMIN_DIAGNOSTIC_NOTIFY_INTERVAL_MS;
  var lastAt = STATE.adminAlertThrottleAtMs[key] || 0;

  if (!CONFIG.ADMIN_DIAGNOSTIC_NOTIFY_ENABLED) {
    return false;
  }

  if (lastAt && now - lastAt < interval) {
    STATE.adminDiagnosticSkipCount += 1;
    return false;
  }

  text =
    "[운영진단]\n" +
    "kind: " + normalizedKind + "\n" +
    "detail: " + safeString(detail) + "\n" +
    buildCompactDiagnosticSnapshot() + "\n" +
    "lastErrorSource: " + safeString(STATE.lastErrorSource) + "\n" +
    "lastError: " + previewText(STATE.lastError, 160);

  STATE.adminAlertThrottleAtMs[key] = now;
  STATE.adminDiagnosticSentCount += 1;
  STATE.adminDiagnosticLastAt = nowIsoString();
  STATE.adminDiagnosticLastKind = normalizedKind;

  try {
    if (!enqueueAsyncTask("notify_admin_diagnostic:" + normalizedKind, function() {
      notifyAdmin(text);
    })) {
      STATE.adminDiagnosticSkipCount += 1;
      logThrottled(
        "error",
        "notify_admin_diagnostic_drop",
        "admin diagnostic enqueue failed kind=" + normalizedKind +
        " detail=" + safeString(detail),
        CONFIG.DIAGNOSTIC_LOG_THROTTLE_MS
      );
      return false;
    }
  } catch (e) {
    STATE.adminDiagnosticSkipCount += 1;
    logThrottled(
      "error",
      "notify_admin_diagnostic_exception",
      "admin diagnostic exception kind=" + normalizedKind +
      " error=" + safeString(e),
      CONFIG.DIAGNOSTIC_LOG_THROTTLE_MS
    );
    return false;
  }

  return true;
}

function recordHttpCall(endpoint, startedAtMs, status, errorText) {
  var finishedAtMs = nowMillis();
  var elapsedMs = finishedAtMs - startedAtMs;
  var key = trimString(endpoint) || "http";
  var stats = STATE.httpStatsByEndpoint[key];
  var finalError = trimString(errorText);

  if (!stats) {
    stats = {
      count: 0,
      lastStatus: "",
      lastElapsedMs: 0,
      lastError: "",
      lastAt: ""
    };
  }

  stats.count += 1;
  stats.lastStatus = String(status);
  stats.lastElapsedMs = elapsedMs;
  stats.lastError = finalError;
  stats.lastAt = nowIsoString();
  STATE.httpStatsByEndpoint[key] = stats;

  STATE.httpCallCount += 1;
  STATE.httpLastEndpoint = key;
  STATE.httpLastStartedAt = safeString(startedAtMs);
  STATE.httpLastFinishedAt = stats.lastAt;
  STATE.httpLastDurationMs = elapsedMs;
  STATE.httpLastStatus = String(status);
  STATE.httpLastError = finalError;

  if (finalError) {
    recordLastError("http:" + key, finalError);
  }

  Log.i(
    "http completed endpoint=" + key +
    " status=" + String(status) +
    " elapsedMs=" + String(elapsedMs) +
    " error=" + (finalError || "-")
  );

  if (finalError) {
    notifyAdminDiagnosticThrottled(
      "http_error",
      "endpoint=" + key +
      " status=" + String(status) +
      " elapsedMs=" + String(elapsedMs) +
      " error=" + finalError
    );
  } else if (elapsedMs >= CONFIG.HTTP_SLOW_LOG_THRESHOLD_MS) {
    notifyAdminDiagnosticThrottled(
      "slow_http",
      "endpoint=" + key +
      " status=" + String(status) +
      " elapsedMs=" + String(elapsedMs)
    );
  }

  return elapsedMs;
}

function sleepMs(ms) {
  try {
    java.lang.Thread.sleep(ms);
  } catch (e) {}
}

function startThread(runFn, threadName) {
  var label = trimString(threadName) || "async-thread";
  var runnable = new java.lang.Runnable({
    run: function() {
      try {
        runFn();
      } catch (e) {
        Log.e("thread crashed name=" + label + " error=" + safeString(e));
      }
    }
  });
  var thread = new java.lang.Thread(runnable);
  try {
    thread.setName(label);
  } catch (e2) {}
  thread.start();
  return thread;
}

function cleanupRuntime(runtime, reason) {
  var normalizedReason = trimString(reason) || "duplicate_boot";

  if (!runtime) {
    return;
  }

  runtime.stopRequested = true;

  try {
    if (runtime.state) {
      runtime.state.pollingEnabled = false;
      runtime.state.pollingSyncQueued = false;
    }
  } catch (e) {}

  try {
    if (bot && typeof bot.removeListener === "function" && runtime.listeners) {
      if (runtime.listeners.message) {
        try {
          bot.removeListener(Event.MESSAGE, runtime.listeners.message);
        } catch (e2) {}
      }
      if (runtime.listeners.command) {
        try {
          bot.removeListener(Event.COMMAND, runtime.listeners.command);
        } catch (e3) {}
      }
    }
  } catch (e4) {}

  Log.i(
    "runtime cleanup requested instanceId=" + safeString(runtime.instanceId) +
    " reason=" + normalizedReason
  );

  notifyAdminDiagnosticThrottled(
    "duplicate_boot",
    "previousInstanceId=" + safeString(runtime.instanceId) +
    " reason=" + normalizedReason +
    " newInstanceId=" + safeString(STATE.instanceId)
  );
}

function getAsyncQueueSize() {
  try {
    if (!STATE.asyncTaskQueue) {
      return 0;
    }
    return STATE.asyncTaskQueue.size();
  } catch (e) {
    return -1;
  }
}

function ensureAsyncWorkersStarted() {
  var i;
  var thread;

  if (STATE.asyncWorkersStarted) {
    return;
  }

  STATE.asyncTaskQueue = new java.util.concurrent.LinkedBlockingQueue(CONFIG.ASYNC_QUEUE_CAPACITY);
  STATE.asyncWorkersStarted = true;
  RUNTIME.workerThreads = [];

  for (i = 0; i < CONFIG.ASYNC_WORKER_COUNT; i++) {
    (function(workerIndex) {
      thread = startThread(
        function() {
          var task;
          var taskLabel;

          STATE.asyncWorkerStartedCount += 1;
          Log.i(
            "async worker started index=" + String(workerIndex) +
            " instanceId=" + safeString(STATE.instanceId)
          );
          while (!RUNTIME.stopRequested) {
            try {
              task = STATE.asyncTaskQueue.poll(1000, java.util.concurrent.TimeUnit.MILLISECONDS);
            } catch (takeError) {
              STATE.asyncLastError =
                "async queue take failed worker=" + String(workerIndex) +
                " error=" + safeString(takeError);
              recordLastError("async_take", STATE.asyncLastError);
              Log.e(STATE.asyncLastError);
              sleepMs(500);
              continue;
            }

            if (!task) {
              continue;
            }

            if (!task || typeof task.run !== "function") {
              continue;
            }

            taskLabel = trimString(task.label) || ("async-task-" + String(workerIndex));
            try {
              task.run();
              STATE.asyncExecutedCount += 1;
              recordLastSuccess("async:" + normalizeTaskLabelForStats(taskLabel));
            } catch (taskError) {
              STATE.asyncLastError =
                "async task failed worker=" + String(workerIndex) +
                " label=" + taskLabel +
                " error=" + safeString(taskError);
              recordLastError("async_task", STATE.asyncLastError);
              Log.e(STATE.asyncLastError);
            }
          }
          Log.i(
            "async worker stopped index=" + String(workerIndex) +
            " instanceId=" + safeString(STATE.instanceId)
          );
        },
        "PpiraeAsyncWorker-" + String(workerIndex)
      );
      RUNTIME.workerThreads.push(thread);
    })(i + 1);
  }

  Log.i(
    "async workers started count=" + String(CONFIG.ASYNC_WORKER_COUNT) +
    " queueCapacity=" + String(CONFIG.ASYNC_QUEUE_CAPACITY) +
    " instanceId=" + safeString(STATE.instanceId)
  );
}

function enqueueAsyncTask(label, runFn) {
  var normalizedLabel = trimString(label) || "async-task";
  var offered = false;
  var queueSize;

  if (typeof runFn !== "function") {
    return false;
  }

  if (RUNTIME.stopRequested) {
    STATE.asyncDroppedCount += 1;
    STATE.asyncEnqueueFailCount += 1;
    STATE.asyncLastError =
      "async enqueue skipped after stop label=" + normalizedLabel;
    recordLastError("async_enqueue_stopped", STATE.asyncLastError);
    logThrottled("error", "async_enqueue_stopped", STATE.asyncLastError, CONFIG.DIAGNOSTIC_LOG_THROTTLE_MS);
    return false;
  }

  ensureAsyncWorkersStarted();

  try {
    offered = STATE.asyncTaskQueue.offer({
      label: normalizedLabel,
      run: runFn
    });
  } catch (e) {
    STATE.asyncEnqueueFailCount += 1;
    STATE.asyncLastError =
      "async enqueue failed label=" + normalizedLabel +
      " error=" + safeString(e);
    recordLastError("async_enqueue_exception", STATE.asyncLastError);
    Log.e(STATE.asyncLastError);
    return false;
  }

  if (!offered) {
    STATE.asyncDroppedCount += 1;
    STATE.asyncEnqueueFailCount += 1;
    STATE.asyncLastError =
      "async queue full label=" + normalizedLabel +
      " size=" + String(getAsyncQueueSize());
    recordLastError("async_queue_full", STATE.asyncLastError);
    Log.e(STATE.asyncLastError);
    return false;
  }

  STATE.asyncEnqueuedCount += 1;
  STATE.asyncLastEnqueueLabel = normalizedLabel;
  updateTaskLabelStats(normalizedLabel);
  queueSize = updateAsyncQueueHighWaterMark();
  if (queueSize >= Math.max(10, Math.floor(CONFIG.ASYNC_QUEUE_CAPACITY * 0.7))) {
    logThrottled(
      "info",
      "async_queue_pressure",
      "async queue pressure size=" + String(queueSize) +
      " highWater=" + String(STATE.asyncQueueHighWaterMark) +
      " lastLabel=" + normalizedLabel,
      CONFIG.DIAGNOSTIC_LOG_THROTTLE_MS
    );
    notifyAdminDiagnosticThrottled(
      "async_queue_pressure",
      "size=" + String(queueSize) +
      " highWater=" + String(STATE.asyncQueueHighWaterMark) +
      " lastLabel=" + normalizedLabel
    );
  }
  return true;
}

function hasCommandPrefix(text) {

}

function containsYouTubeUrl(text) {
  var value = safeString(text);
  if (!value) {
    return false;
  }
  return /(?:youtu\.be\/|youtube\.com\/watch|youtube\.com\/shorts\/|m\.youtube\.com\/watch)/i.test(value);
}

function normalizeMessages(messages) {
  var result = [];
  var i;
  var item;

  if (!messages) {
    return result;
  }

  if (!isArray(messages)) {
    item = trimString(messages);
    if (item) {
      result.push(item);
    }
    return result;
  }

  for (i = 0; i < messages.length; i++) {
    item = trimString(messages[i]);
    if (item) {
      result.push(item);
    }
  }
  return result;
}

function cleanupRecentHandledEvents() {
  var key;
  var entry;
  var ttl;
  var now = nowMillis();
  for (key in STATE.recentHandledEvents) {
    if (!STATE.recentHandledEvents.hasOwnProperty(key)) {
      continue;
    }
    entry = STATE.recentHandledEvents[key];
    ttl = CONFIG.RECENT_EVENT_TTL_MS;
    if (entry && typeof entry === "object") {
      ttl = entry.ttl || ttl;
      entry = entry.at;
    }
    if (now - entry > ttl) {
      delete STATE.recentHandledEvents[key];
      STATE.recentHandledCleanupCount += 1;
    }
  }
}

function wasEventRecentlyHandled(key) {
  var normalized = trimString(key);
  if (!normalized) {
    return false;
  }

function parseCommandText(text) {
  var trimmed = trimString(text);
  var body;
  var tokens;

  if (!hasCommandPrefix(trimmed)) {
    return null;
  }

  body = trimString(trimmed.substring(CONFIG.COMMAND_PREFIX.length));
  if (!body) {
    return null;
  }

  tokens = body.split(/\s+/);
  return {
    command: trimString(tokens[0]),
    args: tokens.slice(1)
  };
}

function parseAdminCommandText(text) {
  var parsed = parseCommandText(text);
  var trimmed = trimString(text);
  var tokens;
  var command;

  if (parsed) {
    return parsed;
  }

  if (!trimmed) {
    return null;
  }

  tokens = trimmed.split(/\s+/);
  command = trimString(tokens[0]);
  if (!isKnownAdminCommand(command)) {
    return null;
  }

  return {
    command: command,
    args: tokens.slice(1)
  };
}

function parseCommandEventPayload(chat) {
  var command = trimString(chat && chat.command);
  var args = toPlainArray(chat && chat.args);

  if (command) {
    return {
      command: command,
      args: args
    };
  }

  return parseCommandText(getMessageText(chat));
}

// ===== 8. Room Cache =====
function addRoomOrder(channelId) {
  var i;
  var evictedChannelId;
  for (i = 0; i < STATE.roomOrder.length; i++) {
    if (STATE.roomOrder[i] === channelId) {
      STATE.roomOrder.splice(i, 1);
      break;
    }
  }
  STATE.roomOrder.unshift(channelId);
  while (STATE.roomOrder.length > CONFIG.MAX_RECENT_ROOMS) {
    evictedChannelId = STATE.roomOrder.pop();
    if (evictedChannelId && STATE.roomsByChannelId[evictedChannelId]) {
      delete STATE.roomsByChannelId[evictedChannelId];
      STATE.roomCacheEvictionCount += 1;
      logThrottled(
        "info",
        "room_cache_evict",
        "room cache evicted channelId=" + safeString(evictedChannelId) +
        " roomOrder=" + String(STATE.roomOrder.length) +
        " cachedRooms=" + String(roomCount()),
        CONFIG.DIAGNOSTIC_LOG_THROTTLE_MS
      );
    }
  }
}

function roomCount() {
  var count = 0;
  var key;
  for (key in STATE.roomsByChannelId) {
    if (STATE.roomsByChannelId.hasOwnProperty(key)) {
      count += 1;
    }
  }
  return count;
}

}

function cacheRoom(msg) {
  var channelId = getChannelId(msg);
  var roomName = getRoomName(msg);
  var item;
  var isNew = false;

  if (!channelId || !roomName) {
    return null;
  }

  item = STATE.roomsByChannelId[channelId];
  if (!item) {
    item = {};
    isNew = true;
  }

  item.channelId = channelId;
  item.room = roomName;
  item.isGroupChat = getIsGroupChat(msg);
  item.packageName = getPackageName(msg) || getKakaoPackageName();
  item.lastAuthor = getAuthorName(msg);
  item.lastLogId = getLogId(msg);
  item.lastSeenAt = nowIsoString();

  STATE.roomsByChannelId[channelId] = item;
  addRoomOrder(channelId);

  if (isNew && CONFIG.NOTIFY_ADMIN_ON_NEW_ROOM && item.room !== CONFIG.ADMIN_ROOM_NAME) {
    notifyAdminAsync(
      "[새 방 감지]\n" +
      "room: " + item.room + "\n" +
      "channelId: " + item.channelId + "\n" +
      "isGroupChat: " + String(item.isGroupChat)
    );
  }

  return item;
}

function getRecentRoomsText(limit) {
  var lines = [];
  var i;
  var item;
  var max = limit || 15;

  for (i = 0; i < STATE.roomOrder.length; i++) {
    if (lines.length >= max) {
      break;
    }
    item = STATE.roomsByChannelId[STATE.roomOrder[i]];
    if (!item) {
      continue;
    }
    lines.push(buildRoomLine(item, lines.length));
  }

  if (lines.length === 0) {
    return "캐시된 방이 없다.";
  }
  return "[최근 감지된 방]\n" + lines.join("\n");
}

function searchRoomsText(keyword) {
  var lines = [];
  var i;
  var item;
  var lower = trimString(keyword).toLowerCase();

  if (!lower) {
    return "검색어가 비어 있다.";
  }

  for (i = 0; i < STATE.roomOrder.length; i++) {
    item = STATE.roomsByChannelId[STATE.roomOrder[i]];
    if (!item) {
      continue;
    }
    if (
      safeString(item.room).toLowerCase().indexOf(lower) >= 0 ||
      safeString(item.channelId).indexOf(lower) >= 0
    ) {
      lines.push(buildRoomLine(item, lines.length));
    }
  }

  if (lines.length === 0) {
    return "검색 결과가 없다: " + keyword;
  }
  return "[검색 결과]\n" + lines.join("\n");
}

function findRoomItem(keywordOrChannelId) {
  var keyword = trimString(keywordOrChannelId);
  var i;
  var item;
  var lower;

  if (!keyword) {
    return null;
  }

  if (STATE.roomsByChannelId[keyword]) {
    return STATE.roomsByChannelId[keyword];
  }

  for (i = 0; i < STATE.roomOrder.length; i++) {
    item = STATE.roomsByChannelId[STATE.roomOrder[i]];
    if (item && safeString(item.room) === keyword) {
      return item;
    }
  }

  lower = keyword.toLowerCase();
  for (i = 0; i < STATE.roomOrder.length; i++) {
    item = STATE.roomsByChannelId[STATE.roomOrder[i]];
    if (item && safeString(item.room).toLowerCase().indexOf(lower) >= 0) {
      return item;
    }
  }

  return null;
}

// ===== 9. Send Helpers =====
function botSendRoom(roomName, text, packageName) {
  var resolvedRoomName = trimString(roomName);
  var resolvedPackage = trimString(packageName) || getKakaoPackageName();
  var result = false;

  if (!resolvedRoomName || !trimString(text)) {
    return false;
  }

  try {
    result = bot.send(resolvedRoomName, text, resolvedPackage);
    if (result === false) {
      Log.e(
        "bot.send returned false room=" + resolvedRoomName +
        " package=" + resolvedPackage
      );
      return false;
    }
    return true;
  } catch (e) {
    Log.e(
      "bot.send failed room=" + resolvedRoomName +
      " package=" + resolvedPackage +
      " error=" + safeString(e)
    );
    return false;
  }
}

function sendMessagesToRoom(roomName, messages, packageName, traceId) {
  var normalized = normalizeMessages(messages);
  var i;
  var ok = true;
  var sentCount = 0;
  var lastError = "";
  var startedAtMs = nowMillis();

  for (i = 0; i < normalized.length; i++) {
    STATE.sendAttemptCount += 1;
    if (!botSendRoom(roomName, normalized[i], packageName)) {
      ok = false;
      STATE.sendFailureCount += 1;
      lastError =
        "bot.send failed at index=" + String(i + 1) +
        " traceId=" + safeString(traceId);
      STATE.sendLastError = lastError;
      recordLastError("send", lastError);
      break;
    }
    sentCount += 1;
    if (i < normalized.length - 1) {
      sleepMs(CONFIG.SEND_DELAY_MS);
    }
  }

  STATE.sendLastDurationMs = nowMillis() - startedAtMs;
  if (ok) {
    STATE.sendLastSuccessAt = nowIsoString();
    recordLastSuccess("send");
  }

  return {
    ok: ok,
    sentCount: sentCount,
    error: ok ? "" : lastError
  };
}

function httpPostJson(url, payload, options) {
  var response;
  var body;
  var parsed;
  var endpoint = trimString(options && options.endpoint) || "http_post";
  var timeoutMs = parseInt(options && options.timeoutMs, 10);
  var startedAtMs = nowMillis();
  var elapsedMs;

  if (!(timeoutMs > 0)) {
    timeoutMs = CONFIG.HTTP_TIMEOUT_MS;
  }

  try {
    response = org.jsoup.Jsoup.connect(url)
      .ignoreContentType(true)
      .ignoreHttpErrors(true)
      .header("Content-Type", "application/json; charset=UTF-8")
      .header("Accept", "application/json")
      .requestBody(JSON.stringify(payload))
      .method(org.jsoup.Connection.Method.POST)
      .timeout(timeoutMs)
      .execute();

    body = response.body();
    parsed = null;
    try {
      parsed = JSON.parse(body);
    } catch (parseError) {}

    elapsedMs = recordHttpCall(endpoint, startedAtMs, response.statusCode(), "");

    return {
      ok: response.statusCode() >= 200 && response.statusCode() < 300,
      status: response.statusCode(),
      body: body,
      json: parsed,
      error: "",
      elapsedMs: elapsedMs
    };
  } catch (e) {
    elapsedMs = recordHttpCall(endpoint, startedAtMs, -1, safeString(e));
    return {
      ok: false,
      status: -1,
      body: "",
      json: null,
      error: safeString(e),
      elapsedMs: elapsedMs
    };
  }
}

function handleMessageEventResponse(msg, response) {
  var data;
  var messages;
  var delivery;
  var deliveryVia;
  var traceId;

  if (!response.ok || !response.json) {
    notifyAdminAsync(
      "[이벤트전달실패]\n" +
      "room: " + getRoomName(msg) + "\n" +
      "channelId: " + getChannelId(msg) + "\n" +
      "author: " + getAuthorName(msg) + "\n" +
      "logId: " + getLogId(msg) + "\n" +
      "status: " + String(response.status) + "\n" +
      "error: " + safeString(response.error)
    );
    return;
  }

  data = response.json;
  if (!data.ok) {
    notifyAdminAsync(
      "[이벤트업무오류]\n" +
      "room: " + getRoomName(msg) + "\n" +
      "channelId: " + getChannelId(msg) + "\n" +
      "author: " + getAuthorName(msg) + "\n" +
      "logId: " + getLogId(msg) + "\n" +
      "error: " + safeString(data.error)
    );
    return;
  }

  traceId = trimString(data.trace_id) || trimString(getLogId(msg)) || nowIsoString();
  messages = normalizeMessages(data.messages);
  delivery = data.meta && typeof data.meta === "object" ? data.meta.delivery : null;
  deliveryVia = trimString(delivery && delivery.via);

  if (deliveryVia === "polling_outbox") {
    triggerPollingSyncNow("event_response_queued");
  }

  if (messages.length > 0) {
    sendMessagesToCurrentRoomAsync(msg, messages, traceId);
  }
}

function topTaskLabelSummary(limit) {
  var entries = [];
  var key;
  var i;
  var item;
  var max = limit || 8;

  for (key in STATE.taskLabelCounts) {
    if (STATE.taskLabelCounts.hasOwnProperty(key)) {
      entries.push({
        label: key,
        count: STATE.taskLabelCounts[key]
      });
    }
  }

  entries.sort(function(a, b) {
    return b.count - a.count;
  });

  if (entries.length === 0) {
    return "-";
  }

  var parts = [];
  for (i = 0; i < entries.length && i < max; i++) {
    item = entries[i];
    parts.push(item.label + "=" + String(item.count));
  }
  return parts.join(", ");
}

function notifyPollingError(text) {
  var normalized = safeString(text);
  var now = nowMillis();

  if (!CONFIG.POLLING_NOTIFY_ADMIN) {
    return;
  }

  if (
    STATE.pollingLastErrorNotifiedAtMs &&
    now - STATE.pollingLastErrorNotifiedAtMs < CONFIG.POLLING_ERROR_NOTIFY_INTERVAL_MS
  ) {
    return;
  }

  STATE.pollingLastErrorNotifiedAtMs = now;
  notifyAdminDiagnosticThrottled("polling_error", normalized);
}

function normalizeOutboxItems(data) {
  var meta = data && data.meta ? data.meta : null;
  var rawItems = meta && isArray(meta.items) ? meta.items : [];
  var items = [];
  var i;

  for (i = 0; i < rawItems.length; i++) {
    if (rawItems[i] && typeof rawItems[i] === "object") {
      items.push(rawItems[i]);
    }
  }

  return items;
}

function resolveOutboxTarget(item) {
  var roomName = trimString(item && item.room_name);
  var channelId = trimString(item && item.channel_id);
  var cached = null;

  if (channelId && STATE.roomsByChannelId[channelId]) {
    cached = STATE.roomsByChannelId[channelId];
    return {
      roomName: trimString(cached.room),
      channelId: trimString(cached.channelId || channelId),
      packageName: trimString(cached.packageName) || getKakaoPackageName()
    };
  }

  if (!roomName) {
    cached = findRoomItem(channelId || trimString(item && item.room_key));
    if (cached) {
      roomName = trimString(cached.room);
      channelId = trimString(cached.channelId || channelId);
    }
  }

  if (!roomName) {
    return null;
  }

  return {
    roomName: roomName,
    channelId: channelId,
    packageName: trimString(cached && cached.packageName) || getKakaoPackageName()
  };
}

function extractOutboxMessages(item) {
  if (isArray(item && item.messages)) {
    return normalizeMessages(item.messages);
  }
  return normalizeMessages(item && item.message);
}

function ackOutboxItem(messageId, success, incrementRetry, detail) {
  var response = postPollingAck([messageId], success, incrementRetry);
  if (!response.ok || !response.json || !response.json.ok) {
    STATE.pollingLastError =
      "ack failed id=" + String(messageId) +
      " success=" + String(!!success) +
      " detail=" + safeString(detail) +
      " status=" + String(response.status) +
      " error=" + safeString(response.error);
    recordLastError("polling_ack", STATE.pollingLastError);
    notifyPollingError(STATE.pollingLastError);
    return false;
  }
  return true;
}

function runSinglePollingCycle(triggerReason) {
  var response;
  var data;
  var items;
  var i;
  var item;
  var target;
  var messages;
  var sendResult;
  var ackOk;
  var sentCount = 0;
  var ackSuccessCount = 0;
  var ackFailCount = 0;
  var startedAtMs = nowMillis();
  var elapsedMs = 0;
  var normalizedTrigger = trimString(triggerReason) || "manual";

  if (!CONFIG.POLLING_ENABLED) {
    return {
      ok: false,
      skipped: true,
      reason: "polling disabled"
    };
  }

  if (STATE.pollingBusy) {
    return {
      ok: false,
      skipped: true,
      reason: "polling busy"
    };
  }

  STATE.pollingBusy = true;
  STATE.pollingCycleCount += 1;
  STATE.pollingLastPolledAt = nowIsoString();
  STATE.pollingLastCycleAt = STATE.pollingLastPolledAt;
  STATE.pollingLastTrigger = normalizedTrigger;
  STATE.pollingLastError = "";
  Log.i(
    "polling cycle start trigger=" + normalizedTrigger +
    " cycle=" + String(STATE.pollingCycleCount) +
    " queueSize=" + String(getAsyncQueueSize())
  );

  try {
    response = postPollingPull(CONFIG.POLLING_LIMIT);
    if (!response.ok || !response.json) {
      STATE.pollingLastError =
        "pull failed trigger=" + normalizedTrigger +
        " status=" + String(response.status) +
        " error=" + safeString(response.error);
      recordLastError("polling_pull", STATE.pollingLastError);
      notifyPollingError(STATE.pollingLastError);
      return {
        ok: false,
        pulled: 0,
        sent: 0,
        error: STATE.pollingLastError
      };
    }

    data = response.json;
    if (!data.ok) {
      STATE.pollingLastError =
        "pull business error trigger=" + normalizedTrigger +
        " error=" + safeString(data.error);
      recordLastError("polling_pull_business", STATE.pollingLastError);
      notifyPollingError(STATE.pollingLastError);
      return {
        ok: false,
        pulled: 0,
        sent: 0,
        error: STATE.pollingLastError
      };
    }

    items = normalizeOutboxItems(data);
    STATE.pollingLastPulledCount = items.length;

    for (i = 0; i < items.length; i++) {
      item = items[i];
      target = resolveOutboxTarget(item);
      messages = extractOutboxMessages(item);

      if (!target || !target.roomName || messages.length === 0) {
        ackOk = ackOutboxItem(
          item.id,
          false,
          CONFIG.POLLING_ACK_INCREMENT_RETRY_ON_FAILURE,
          "invalid target or empty messages"
        );
        if (ackOk) {
          ackFailCount += 1;
        }
        continue;
      }

      sendResult = sendMessagesToRoom(
        target.roomName,
        messages,
        target.packageName,
        trimString(item.trace_id) || ("outbox-" + String(item.id))
      );

      if (sendResult.ok) {
        sentCount += sendResult.sentCount;
      }

      ackOk = ackOutboxItem(
        item.id,
        sendResult.ok,
        CONFIG.POLLING_ACK_INCREMENT_RETRY_ON_FAILURE && !sendResult.ok,
        sendResult.error
      );

      if (ackOk) {
        if (sendResult.ok) {
          ackSuccessCount += 1;
        } else {
          ackFailCount += 1;
        }
      }

      if (!sendResult.ok) {
        STATE.pollingLastError =
          "delivery failed id=" + String(item.id) +
          " room=" + safeString(target.roomName) +
          " error=" + safeString(sendResult.error);
        recordLastError("polling_send", STATE.pollingLastError);
        Log.e(STATE.pollingLastError);
      }
    }

    STATE.pollingLastSuccessAt = nowIsoString();
    STATE.pollingLastSentCount = sentCount;
    STATE.pollingLastAckSuccessCount = ackSuccessCount;
    STATE.pollingLastAckFailCount = ackFailCount;
    elapsedMs = nowMillis() - startedAtMs;
    STATE.pollingLastDurationMs = elapsedMs;
    recordLastSuccess("polling");
    Log.i(
      "polling cycle end trigger=" + normalizedTrigger +
      " ok=true pulled=" + String(items.length) +
      " sent=" + String(sentCount) +
      " ackSuccess=" + String(ackSuccessCount) +
      " ackFail=" + String(ackFailCount) +
      " elapsedMs=" + String(elapsedMs)
    );

    if (elapsedMs >= CONFIG.POLLING_SLOW_LOG_THRESHOLD_MS) {
      notifyAdminDiagnosticThrottled(
        "slow_polling",
        "trigger=" + normalizedTrigger +
        " pulled=" + String(items.length) +
        " sent=" + String(sentCount) +
        " ackSuccess=" + String(ackSuccessCount) +
        " ackFail=" + String(ackFailCount) +
        " elapsedMs=" + String(elapsedMs)
      );
    }

    return {
      ok: true,
      pulled: items.length,
      sent: sentCount,
      ackSuccessCount: ackSuccessCount,
      ackFailCount: ackFailCount,
      elapsedMs: elapsedMs
    };
  } catch (e) {
    STATE.pollingLastError =
      "polling exception trigger=" + normalizedTrigger +
      " error=" + safeString(e);
    recordLastError("polling_exception", STATE.pollingLastError);
    notifyPollingError(STATE.pollingLastError);
    return {
      ok: false,
      pulled: 0,
      sent: 0,
      error: STATE.pollingLastError
    };
  } finally {
    STATE.pollingLastDurationMs = nowMillis() - startedAtMs;
    STATE.pollingBusy = false;
  }
}

function triggerPollingSyncNow(triggerReason) {
  var normalizedTrigger = trimString(triggerReason) || "manual";

  if (!CONFIG.POLLING_ENABLED) {
    return false;
  }

  if (STATE.pollingSyncQueued || STATE.pollingSyncRunning) {
    STATE.pollingSyncSkipCount += 1;
    logThrottled(
      "info",
      "polling_sync_skip",
      "polling sync skipped trigger=" + normalizedTrigger +
      " queued=" + String(STATE.pollingSyncQueued) +
      " running=" + String(STATE.pollingSyncRunning) +
      " busy=" + String(STATE.pollingBusy) +
      " skipCount=" + String(STATE.pollingSyncSkipCount),
      CONFIG.DIAGNOSTIC_LOG_THROTTLE_MS
    );
    notifyAdminDiagnosticThrottled(
      "polling_sync_backpressure",
      "trigger=" + normalizedTrigger +
      " queued=" + String(STATE.pollingSyncQueued) +
      " running=" + String(STATE.pollingSyncRunning) +
      " busy=" + String(STATE.pollingBusy) +
      " skipCount=" + String(STATE.pollingSyncSkipCount)
    );
    return false;
  }

  STATE.pollingSyncQueued = true;

  if (!enqueueAsyncTask("polling_sync:" + normalizedTrigger, function() {
    STATE.pollingSyncQueued = false;
    STATE.pollingSyncRunning = true;
    try {
      runSinglePollingCycle(normalizedTrigger);
    } finally {
      STATE.pollingSyncRunning = false;
    }
  })) {
    STATE.pollingSyncQueued = false;
    handleAsyncDrop("polling_sync:" + normalizedTrigger, "polling sync enqueue failed");
    return false;
  }

  return true;
}

function startPollingLoop() {
  if (!CONFIG.POLLING_ENABLED) {
    return false;
  }
  if (STATE.pollingThreadRunning) {
    return false;
  }

  STATE.pollingEnabled = true;
  STATE.pollingThreadRunning = true;
  STATE.pollingLastStartedAt = nowIsoString();
  STATE.pollingLoopStartCount += 1;

  RUNTIME.pollingThread = startThread(function() {
    var remaining;
    Log.i(
      "polling loop started intervalMs=" + String(CONFIG.POLLING_INTERVAL_MS) +
      " startCount=" + String(STATE.pollingLoopStartCount) +
      " instanceId=" + safeString(STATE.instanceId)
    );
    try {
      while (STATE.pollingEnabled && !RUNTIME.stopRequested) {
        runSinglePollingCycle("loop");
        remaining = Math.max(1000, CONFIG.POLLING_INTERVAL_MS);
        while (STATE.pollingEnabled && !RUNTIME.stopRequested && remaining > 0) {
          sleepMs(Math.min(1000, remaining));
          remaining -= 1000;
        }
      }
    } finally {
      STATE.pollingThreadRunning = false;
      Log.i("polling loop stopped instanceId=" + safeString(STATE.instanceId));
    }
  }, "PpiraePollingLoop");

  return true;
}

function stopPollingLoop() {
  STATE.pollingEnabled = false;
  STATE.pollingSyncQueued = false;
}

function restartPollingLoop() {
  stopPollingLoop();
  startThread(function() {
    var remaining = 5000;
    while (STATE.pollingThreadRunning && remaining > 0) {
      sleepMs(250);
      remaining -= 250;
    }
    startPollingLoop();
  });
}

function ensurePollingLoopStarted() {
  if (!CONFIG.POLLING_AUTO_START) {
    return;
  }
  startPollingLoop();
}

// ===== 13. Admin Commands =====
function handleAdminCommand(msg, parsedCommand) {
  var command = trimString(parsedCommand.command);
  var args = parsedCommand.args || [];
  var keyword;
  var roomKey;
  var item;
  var dedupeKey;
  var dedupeTtl = CONFIG.RECENT_COMMAND_TTL_MS;

  if (!command) {
    return false;
  }

  dedupeKey =
    "cmd:" +
    getChannelId(msg) +
    ":" +
    command.toLowerCase() +
    ":" +
    trimString(args.join(" "));

  if (wasEventRecentlyHandled(dedupeKey)) {
    return false;
  }
  markEventHandled(dedupeKey, dedupeTtl);

  if (command === "도움말" || command.toLowerCase() === "help") {
    replyAdminCommand(msg, buildHelpText());
    return true;
  }

  if (command === "핑" || command.toLowerCase() === "ping") {
    replyAdminCommand(
      msg,
      "[핑]\n" +
      "bot: " + (ACTUAL_BOT_NAME || CONFIG.BOT_NAME) + "\n" +
      "room: " + getRoomName(msg) + "\n" +
      "channelId: " + getChannelId(msg) + "\n" +
      "isGroupChat: " + String(getIsGroupChat(msg))
    );
    return true;
  }

  if (command === "방목록") {
    replyAdminCommand(msg, getRecentRoomsText(20));
    return true;
  }

  if (command === "방검색") {
    keyword = trimString(args.join(" "));
    if (!keyword) {
      replyAdminCommand(msg, "사용법: 방검색 키워드");
      return true;
    }
    replyAdminCommand(msg, searchRoomsText(keyword));
    return true;
  }

  if (command === "yaml") {
    if (!args || args.length < 2) {
      replyAdminCommand(msg, "사용법: yaml room_key 검색어\n예: yaml stock_openchat_alpha 가족");
      return true;
    }
    roomKey = safeString(args[0]);
    keyword = trimString(args.slice(1).join(" "));
    item = findRoomItem(keyword);
    if (!item) {
      replyAdminCommand(msg, "대상 방을 찾지 못했다: " + keyword);
      return true;
    }
    replyAdminCommand(msg, buildYamlSnippet(item, roomKey));
    return true;
  }

  if (command === "연동상태") {
    replyAdminCommand(msg, buildStatusText() + "\n\n" + buildPollingStatusText());
    return true;
  }

  if (command === "폴링상태") {
    replyAdminCommand(msg, buildPollingStatusText());
    return true;
  }

  if (command === "진단상태" || command === "디버그상태") {
    replyAdminCommand(msg, buildDiagnosticStatusText());
    return true;
  }

  if (command === "폴링시작") {
    startPollingLoop();
    triggerPollingSyncNow("admin_start");
    replyAdminCommand(msg, "폴링을 시작했다.\n\n" + buildPollingStatusText());
    return true;
  }

  if (command === "폴링중지") {
    stopPollingLoop();
    replyAdminCommand(msg, "폴링을 중지했다.\n\n" + buildPollingStatusText());
    return true;
  }

  if (command === "폴링재시작") {
    restartPollingLoop();
    triggerPollingSyncNow("admin_restart");
    replyAdminCommand(msg, "폴링을 재시작했다.\n\n" + buildPollingStatusText());
    return true;
  }

  if (command === "동기화" || command === "폴링한번") {
    triggerPollingSyncNow("admin_sync");
    replyAdminCommand(msg, "폴링 동기화를 요청했다.\n\n" + buildPollingStatusText());
    return true;
  }

  if (command === "내방" || command === "채널확인") {
    replyAdminCommand(msg, buildCurrentRoomText(msg));
    return true;
  }

  return false;
}

// ===== 14. Event Handlers =====
function onMessage(chat) {
  var messageText;
  var parsedCommand;
  var startedAtMs = nowMillis();

  try {
    if (RUNTIME.stopRequested) {
      return;
    }

    cacheRoom(chat);
    messageText = getMessageText(chat);

    if (isAdminRoom(chat)) {
      parsedCommand = parseAdminCommandText(messageText);
      if (parsedCommand) {
        handleAdminCommand(chat, parsedCommand);
        return;
      }
    }

    if (shouldForwardToFastApi(chat)) {
      forwardMessageAsync(chat);
    }
  } catch (e) {
    Log.e("onMessage error: " + safeString(e));
    recordLastError("onMessage", safeString(e));
    if (isAdminRoom(chat)) {
      notifyAdminAsync("[메시지이벤트오류]\nerror: " + safeString(e));
    }
  } finally {
    STATE.lastMessageEventDurationMs = nowMillis() - startedAtMs;
  }
}

function onCommand(chat) {
  var parsedCommand;

  try {
    if (RUNTIME.stopRequested) {
      return;
    }

    cacheRoom(chat);
    if (!isAdminRoom(chat)) {
      return;
    }

    parsedCommand = parseCommandEventPayload(chat);
    if (!parsedCommand) {
      return;
    }

    handleAdminCommand(chat, parsedCommand);
  } catch (e) {
    Log.e("onCommand error: " + safeString(e));
    recordLastError("onCommand", safeString(e));
    if (isAdminRoom(chat)) {
      notifyAdminAsync("[명령이벤트오류]\nerror: " + safeString(e));
    }
  }
}

function initializeBotRuntime() {
  if (PREVIOUS_RUNTIME && DUPLICATE_BOOT_ATTEMPT) {
    STATE.duplicateBootDetected = true;
    STATE.previousInstanceId = safeString(PREVIOUS_RUNTIME.instanceId);
    cleanupRuntime(PREVIOUS_RUNTIME, "duplicate_boot_attempt");
    Log.i(
      "duplicate boot detected previousInstanceId=" + safeString(PREVIOUS_RUNTIME.instanceId) +
      " newInstanceId=" + safeString(STATE.instanceId)
    );
  } else {
    Log.i("fresh boot instanceId=" + safeString(STATE.instanceId));
  }

  if (RUNTIME.initialized) {
    Log.i("runtime already initialized instanceId=" + safeString(STATE.instanceId));
    return;
  }

  try {
    bot.setCommandPrefix(CONFIG.COMMAND_PREFIX);
  } catch (prefixError) {
    recordLastError("setCommandPrefix", safeString(prefixError));
    Log.e("failed to set command prefix error=" + safeString(prefixError));
  }

  try {
    bot.addListener(Event.MESSAGE, onMessage);
    RUNTIME.listeners.message = onMessage;
  } catch (messageListenerError) {
    recordLastError("addListener:message", safeString(messageListenerError));
    Log.e("failed to add message listener error=" + safeString(messageListenerError));
  }

  try {
    bot.addListener(Event.COMMAND, onCommand);
    RUNTIME.listeners.command = onCommand;
  } catch (commandListenerError) {
    recordLastError("addListener:command", safeString(commandListenerError));
    Log.e("failed to add command listener error=" + safeString(commandListenerError));
  }

  ensurePollingLoopStarted();
  ensureAsyncWorkersStarted();
  RUNTIME.initialized = true;

  Log.i("==================");
  Log.i((ACTUAL_BOT_NAME || CONFIG.BOT_NAME) + " loaded");
  Log.i("instanceId=" + safeString(STATE.instanceId));
  Log.i("startupCount=" + String(STATE.startupCount));
  Log.i("API_BASE_URL=" + CONFIG.API_BASE_URL);
  Log.i("DELIVERY_MODE=polling_outbox");
  Log.i("POLLING_AUTO_START=" + String(CONFIG.POLLING_AUTO_START));
  Log.i("POLLING_INTERVAL_MS=" + String(CONFIG.POLLING_INTERVAL_MS));
  Log.i("ASYNC_WORKER_COUNT=" + String(CONFIG.ASYNC_WORKER_COUNT));
  Log.i("ASYNC_QUEUE_CAPACITY=" + String(CONFIG.ASYNC_QUEUE_CAPACITY));
  Log.i("==================");

  if (CONFIG.STARTUP_NOTIFY_ADMIN) {
    notifyAdminAsync(
      "[삐래봇 로드됨]\n" +
      "bot: " + (ACTUAL_BOT_NAME || CONFIG.BOT_NAME) + "\n" +
      "instanceId: " + safeString(STATE.instanceId) + "\n" +
      "startupCount: " + String(STATE.startupCount) + "\n" +
      "duplicateBootDetected: " + String(STATE.duplicateBootDetected) + "\n" +
      "adminRoom: " + CONFIG.ADMIN_ROOM_NAME + "\n" +
      "adminChannelId: " + CONFIG.ADMIN_CHANNEL_ID + "\n" +
      "api: " + CONFIG.API_BASE_URL + "\n" +
      "mode: polling_outbox"
    );
  }
}
